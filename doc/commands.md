Commands:

telebook
    -p Работаем с пользователями
        -all Получаем список всех пользователей
        -add first_name last_name num1 num2 num3 Добавить пользователя
        -fnd id_person || last_name num Находим пользователя по ID или фамилии и номера телефона
        -del id_person удаление пользователя по ID
        -upt id_person first_name last_name num1 num2 num3 обновление данных пользователя
    -n Работаем с номерами
        -add num id_person Добавление нмера пользователю
        -del num Удаление номера
